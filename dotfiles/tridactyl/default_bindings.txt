" ============================================================================
" VISUAL MODE
"
" Visual mode inherits normal mode.
" ============================================================================

bind --mode=visual <Escape> composite js tri.dom.getSelection().empty(); mode normal; hidecmdline
bind --mode=visual <C-[> composite js tri.dom.getSelection().empty(); mode normal ; hidecmdline

bind --mode=visual y composite js tri.dom.getSelection().toString() | clipboard yank

bind --mode=visual # composite js document.location + '#:~:text=' + encodeURIComponent(tri.dom.getSelection().toString()) | clipboard yank

bind --mode=visual s composite js tri.dom.getSelection().toString() | fillcmdline open search
bind --mode=visual S composite js tri.dom.getSelection().toString() | fillcmdline tabopen search

bind --mode=visual l js -d@ const sel = tri.dom.getSelection(); for (let i = 0; i < (JS_ARGS[1] || 1); i++) tri.visual.extendByCharacter(sel, "forward"); @

bind --mode=visual h js -d@ const sel = tri.dom.getSelection(); for (let i = 0; i < (JS_ARGS[1] || 1); i++) tri.visual.extendByCharacter(sel, "backward"); @

bind --mode=visual e js -d@ for (let i = 0; i < (JS_ARGS[1] || 1); i++) tri.dom.getSelection().modify("extend","forward","word") @

bind --mode=visual w js -d@ for (let i = 0; i < (JS_ARGS[1] || 1); i++) tri.visual.extendByWord(tri.dom.getSelection()) @

bind --mode=visual b js -d@ let s=tri.dom.getSelection(); for (let i = 0; i < (JS_ARGS[1] || 1); i++) { s.modify("extend","backward","character"); s.modify("extend","backward","word"); s.modify("extend","forward","character"); } @

bind --mode=visual j js -d@ for (let i = 0; i < (JS_ARGS[1] || 1); i++) tri.dom.getSelection().modify("extend","forward","line") @

bind --mode=visual k js -d@ for (let i = 0; i < (JS_ARGS[1] || 1); i++) tri.dom.getSelection().modify("extend","backward","line") @

bind --mode=visual q composite js tri.dom.getSelection().toString() | text2qr --timeout 5

bind --mode=visual $ js tri.dom.getSelection().modify("extend","forward","lineboundary")
bind --mode=visual 0 js tri.dom.getSelection().modify("extend","backward","lineboundary")

bind --mode=visual = js -d@ let s = tri.dom.getSelection(); for (let i = 0; i < (JS_ARGS[1] || 1); i++) { let n = s.anchorNode.parentNode; let r = n.ownerDocument.createRange(); s.removeAllRanges(); r.selectNodeContents(n); s.addRange(r); } @

bind --mode=visual o js tri.visual.reverseSelection(tri.dom.getSelection())
